cmake_minimum_required(VERSION 3.15)
project(portuNLP LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(portunlp_core STATIC src/cpp/tokenizer.cpp src/cpp/corpus.cpp)
target_include_directories(portunlp_core PUBLIC include)

if(MINGW)
  add_link_options(-static-libgcc -static-libstdc++)
endif()

add_executable(test_tokenizer tests/cpp/test_tokenizer.cpp)
target_link_libraries(test_tokenizer PRIVATE portunlp_core)
add_executable(test_tokenizer_empty tests/cpp/test_tokenizer_empty.cpp)
target_link_libraries(test_tokenizer_empty PRIVATE portunlp_core)
add_executable(test_tokenizer_punct tests/cpp/test_tokenizer_punct.cpp)
target_link_libraries(test_tokenizer_punct PRIVATE portunlp_core)
add_executable(test_tokenizer_unicode tests/cpp/test_tokenizer_unicode.cpp)
target_link_libraries(test_tokenizer_unicode PRIVATE portunlp_core)

find_package(pybind11 CONFIG QUIET)
if(pybind11_FOUND)
  pybind11_add_module(_portunlp_cpp src/cpp/bindings.cpp)
  target_link_libraries(_portunlp_cpp PRIVATE portunlp_core)
  target_include_directories(_portunlp_cpp PRIVATE include)
  install(TARGETS _portunlp_cpp DESTINATION portunlp)
endif()

enable_testing()
add_test(NAME tokenizer COMMAND $<TARGET_FILE:test_tokenizer>)
add_test(NAME tokenizer_empty COMMAND $<TARGET_FILE:test_tokenizer_empty>)
add_test(NAME tokenizer_punct COMMAND $<TARGET_FILE:test_tokenizer_punct>)
add_test(NAME tokenizer_unicode COMMAND $<TARGET_FILE:test_tokenizer_unicode>)
